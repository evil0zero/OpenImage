function(source_root name sentinel)
    set(path "${PROJECT_SOURCE_DIR}/${name}")
    if(NOT EXISTS "${path}")
        set(path "${PROJECT_SOURCE_DIR}/vendor/${name}")
    endif()
    if(NOT EXISTS "${path}/${sentinel}")
        message(FATAL_ERROR "Missing compatible ${name} source: expected ${path}/${sentinel}")
    endif()
    set(${name}_root "${path}" PARENT_SCOPE)
    message(STATUS "${name} source: ${path}")
endfunction()

    target_sources(oi_blake3 PRIVATE "${blake3_root}/c/${name}.c")

endforeach()

option(OPENIMAGE_QT "Build optional Qt desktop interface" OFF)
option(OPENIMAGE_REQUIRE_STATIC_QT "Require static Qt libraries" OFF)
if(OPENIMAGE_QT)
    if(OPENIMAGE_REQUIRE_STATIC_QT)
        find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent PATHS "${Qt6_DIR}" NO_DEFAULT_PATH)
    else()
        find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent)
    endif()
    if(OPENIMAGE_REQUIRE_STATIC_QT)
        get_target_property(qt_core_type Qt6::Core TYPE)
        if(NOT qt_core_type STREQUAL "STATIC_LIBRARY")
            message(FATAL_ERROR "OpenImage's bundled desktop build requires static Qt.")
        endif()
    endif()
    qt_add_executable(openimage-desktop src/desktop.cpp)
    target_link_libraries(openimage-desktop PRIVATE openimage_core Qt6::Widgets Qt6::Concurrent)
    install(TARGETS openimage-desktop RUNTIME DESTINATION bin)
endif()
